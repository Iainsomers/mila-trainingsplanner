function flexAlternativeLoads(text, fallbackZone, storedSeconds, durationForText) { const parts = []; const value = text || ''; let depth = 0, start = 0; for (let i = 0; i < value.length; i++) { if (value[i] === '(') depth++; else if (value[i] === ')') depth = Math.max(0, depth - 1); else if (depth === 0 && (value[i] === '+' || value.slice(i, i + 2) === '//')) { const part = value.slice(start, i).trim(); if (part) parts.push(part); if (value[i] === '/') i++; start = i + 1; } } const last = value.slice(start).trim(); if (last) parts.push(last); return (parts.length ? parts : ['']).flatMap(part => { const range = part.match(/\bz\s*([1-6])\s*(?:>|-)\s*z?\s*([1-6])\b/i); let zones = []; if (range) { const start = Number(range[1]), end = Number(range[2]), step = end >= start ? 1 : -1; for (let z = start; z !== end + step; z += step) zones.push(String(z)); } else { const match = part.match(/\bz\s*([1-6])\b/i); zones = [match ? match[1] : fallbackZone]; } const seconds = durationForText(part) || (parts.length <= 1 ? storedSeconds : 0); return seconds > 0 && zones.every(z => ['1','2','3'].includes(z)) ? zones.map(zone => ({kind:'alt', zone:zone, seconds:seconds/zones.length})) : []; }); }
